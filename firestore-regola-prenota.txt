// Da AGGIUNGERE alle regole Firestore (Firebase console > Firestore > Regole),
// dentro "match /databases/{database}/documents { ... }", senza togliere le regole che ci sono gia'.
// Permette a chiunque di CREARE solo una richiesta di prenotazione web (non leggere, non modificare).
    match /prenotazioni/{id} {
      allow create: if request.auth == null
        && request.resource.data.keys().hasOnly(['nome','nomeSolo','cognomeSolo','telefono','email','clienteId','data','ora','turno','persone','durata','note','stato','canale','causale','consensi','creatoIl','provenienza','fonte'])
        && request.resource.data.stato == 'richiesta'
        && request.resource.data.canale == 'web'
        && request.resource.data.persone is int && request.resource.data.persone >= 1 && request.resource.data.persone <= 60
        && request.resource.data.nome is string && request.resource.data.nome.size() <= 80
        && request.resource.data.telefono is string && request.resource.data.telefono.matches('^\\+393[0-9]{8,9}$')
        && request.resource.data.clienteId == 't' + request.resource.data.telefono.replace('\\+','')
        && request.resource.data.data is string && request.resource.data.data.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && request.resource.data.ora is string && request.resource.data.ora.matches('^[0-9]{2}:[0-9]{2}$')
        && request.resource.data.note is string && request.resource.data.note.size() <= 400
        && request.resource.data.provenienza is string && request.resource.data.provenienza.size() <= 30
        && request.resource.data.fonte is string && request.resource.data.fonte.size() <= 30
        && request.resource.data.creatoIl == request.time;
    }
